ELI5: 排序算法是如何工作的？

通过"身高排队"的类比，理解三种经典排序算法

🎓 生活类比：学生按身高排队

想象一下，老师让全班同学按身高从矮到高排成一队。有三种不同的方法可以做到：

  • 冒泡排序（Bubble Sort）：相邻的两个同学比较身高，如果前面的比后面的高，就交换位置。重复这个过程，直到所有人都排好序。
  • 快速排序（Quick Sort）：选一个同学作为"基准"，比他矮的站左边，比他高的站右边。然后对左右两组分别重复这个过程。
  • 归并排序（Merge Sort）：把班级分成两半，再把每一半继续分下去，直到每组只有一个人。然后把相邻的小组合并，合并时按身高排序。

C++ 实现

// 冒泡排序
void bubbleSort(int arr[], int n) {
    for (int i = 0; i < n - 1; i++) {
        bool swapped = false;
        for (int j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                swap(arr[j], arr[j + 1]);
                swapped = true;
            }
        }
        if (!swapped) break; // 优化：提前终止
    }
}

// 快速排序
int partition(int arr[], int low, int high) {
    int pivot = arr[high];
    int i = low - 1;
    
    for (int j = low; j < high; j++) {
        if (arr[j] <= pivot) {
            i++;
            swap(arr[i], arr[j]);
        }
    }
    swap(arr[i + 1], arr[high]);
    return i + 1;
}

void quickSort(int arr[], int low, int high) {
    if (low < high) {
        int pi = partition(arr, low, high);
        quickSort(arr, low, pi - 1);
        quickSort(arr, pi + 1, high);
    }
}

// 归并排序
void merge(int arr[], int l, int m, int r) {
    int n1 = m - l + 1, n2 = r - m;
    vector<int> L(n1), R(n2);
    
    for (int i = 0; i < n1; i++) L[i] = arr[l + i];
    for (int j = 0; j < n2; j++) R[j] = arr[m + 1 + j];
    
    int i = 0, j = 0, k = l;
    while (i < n1 && j < n2) {
        if (L[i] <= R[j]) arr[k++] = L[i++];
        else arr[k++] = R[j++];
    }
    while (i < n1) arr[k++] = L[i++];
    while (j < n2) arr[k++] = R[j++];
}

void mergeSort(int arr[], int l, int r) {
    if (l < r) {
        int m = l + (r - l) / 2;
        mergeSort(arr, l, m);
        mergeSort(arr, m + 1, r);
        merge(arr, l, m, r);
    }
}

Java 实现

// 冒泡排序
public static void bubbleSort(int[] arr) {
    int n = arr.length;
    for (int i = 0; i < n - 1; i++) {
        boolean swapped = false;
        for (int j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
                swapped = true;
            }
        }
        if (!swapped) break; // 优化：提前终止
    }
}

// 快速排序
private static int partition(int[] arr, int low, int high) {
    int pivot = arr[high];
    int i = low - 1;
    
    for (int j = low; j < high; j++) {
        if (arr[j] <= pivot) {
            i++;
            int temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;
        }
    }
    int temp = arr[i + 1];
    arr[i + 1] = arr[high];
    arr[high] = temp;
    return i + 1;
}

public static void quickSort(int[] arr, int low, int high) {
    if (low < high) {
        int pi = partition(arr, low, high);
        quickSort(arr, low, pi - 1);
        quickSort(arr, pi + 1, high);
    }
}

// 归并排序
private static void merge(int[] arr, int l, int m, int r) {
    int n1 = m - l + 1, n2 = r - m;
    int[] L = new int[n1];
    int[] R = new int[n2];
    
    for (int i = 0; i < n1; i++) L[i] = arr[l + i];
    for (int j = 0; j < n2; j++) R[j] = arr[m + 1 + j];
    
    int i = 0, j = 0, k = l;
    while (i < n1 && j < n2) {
        if (L[i] <= R[j]) arr[k++] = L[i++];
        else arr[k++] = R[j++];
    }
    while (i < n1) arr[k++] = L[i++];
    while (j < n2) arr[k++] = R[j++];
}

public static void mergeSort(int[] arr, int l, int r) {
    if (l < r) {
        int m = l + (r - l) / 2;
        mergeSort(arr, l, m);
        mergeSort(arr, m + 1, r);
        merge(arr, l, m, r);
    }
}

何时使用哪种算法？

🫧 冒泡排序

  • 小规模数据集（n < 50）
  • 教学和学习目的
  • 数据已经基本有序
  • 需要简单稳定的实现
  • 内存受限的环境

⚡ 快速排序

  • 通用场景的首选
  • 内存排序（原地排序）
  • 平均性能至关重要
  • 不需要稳定排序
  • 大数据集处理

🔗 归并排序

  • 需要稳定排序
  • 外部排序（大文件）
  • 链表排序
  • 需要保证 O(n log n)
  • 并行化处理

常见误区

❌ "冒泡排序总是很慢" ✅ 实际上，对于已经排好序的数据，冒泡排序只需要 O(n) 时间！通过添加"是否发生交换"的标志位，可以在检测到数组已有序时提前终止。
❌ "快速排序总是最快的" ✅ 快速排序的最坏情况是 O(n²)，当每次选择的基准都是最大或最小值时会出现。实际性能取决于基准选择策略和数据分布。
❌ "归并排序需要额外空间是个缺点" ✅ 虽然归并排序需要 O(n) 额外空间，但它提供了稳定性保证，并且在处理链表和外部排序时表现优异。这是用空间换时间的典型例子。
